邮件

Email 可以用 gmail，一个手机注册两个号，手机上清掉浏览数据开梯子注册成功率更高。

最后，下载那个网页应用 gmail，设置开机自启动，然后设置里面打开 Windows 通知就可以享受到邮件提醒了。手机上下载 gmail 不开梯子也能收到提醒。

音乐

平台版权太烦了，Spotify 本地播放即可。

网易云的下载下来，可以用 AthBe1337/ncmdump-web: 适用于Web页面的网易云加密格式NCM解密工具，直接在前端转换，无需后端交互。 这个工具。直接访问 NCM 解密工具 即可。然后导入到 Spotify 里就好了。

QQ 音乐有点麻烦，首先得下载到本地，获得 .mflac 格式，然后用 Acooldog/QQMusic-mflac-to-flac 这个工具转成 .flac。然后你导入 Spotify 会发现缺失了专辑图片，很丑。为了解决这个问题，我们先运行 clear_images.py：

import os
from mutagen.flac import FLAC
from mutagen.id3 import ID3

def clear_flac(path):
    try:
        audio = FLAC(path)
        count = len(audio.pictures)
        audio.clear_pictures()
        audio.save()
        print(f"[FLAC] 已清除 {count} 个封面 -> {path}")
    except Exception as e:
        print(f"[FLAC] 失败 -> {path}  |  {e}")

def clear_mp3(path):
    try:
        audio = ID3(path)
        count = len(audio.getall("APIC"))
        audio.delall("APIC")
        audio.save()
        print(f"[MP3] 已清除 {count} 个封面 -> {path}")
    except Exception as e:
        print(f"[MP3] 失败 -> {path}  |  {e}")

def scan_folder(folder):
    for root, dirs, files in os.walk(folder):
        for file in files:
            full_path = os.path.join(root, file)
            if file.lower().endswith(".flac"):
                clear_flac(full_path)
            elif file.lower().endswith(".mp3"):
                clear_mp3(full_path)

if __name__ == "__main__":
    folder = "C:\\Users\\Administrator\\Music"
    scan_folder(folder)
    print("全部处理完成")

import os
import time
import requests

from mutagen.flac import FLAC, Picture

from mutagen.mp3 import MP3
from mutagen.id3 import ID3, APIC, ID3NoHeaderError


MUSIC_DIR = "C:\\Users\\Administrator\\Music"


def search_qq_album_mid(artist, title):
    url = "https://c.y.qq.com/soso/fcgi-bin/client_search_cp"
    params = {
        "w": f"{artist} {title}",
        "format": "json",
        "p": 1,
        "n": 1
    }
    headers = {
        "Referer": "https://y.qq.com/",
        "User-Agent": "Mozilla/5.0"
    }

    r = requests.get(url, params=params, headers=headers, timeout=10)
    data = r.json()

    try:
        return data["data"]["song"]["list"][0]["albummid"]
    except Exception:
        return None


def download_cover(album_mid):
    url = f"https://y.qq.com/music/photo_new/T002R300x300M000{album_mid}_2.jpg?max_age=2592000"
    r = requests.get(url, timeout=15)

    print("  状态:", r.status_code)
    print("  类型:", r.headers.get("Content-Type"))
    print("  大小:", len(r.content))

    if (
        r.status_code == 200
        and "image" in str(r.headers.get("Content-Type"))
        and len(r.content) > 1000
    ):
        return r.content
    return None


def embed_cover_flac(flac_path, image_data):
    audio = FLAC(flac_path)
    audio.clear_pictures()

    picture = Picture()
    picture.data = image_data
    picture.type = 3  # 3 = Cover (front)；你之前写 0 也能用，但 3 更标准
    picture.mime = "image/jpeg"
    picture.desc = "Cover"

    audio.add_picture(picture)
    audio.save()

    audio2 = FLAC(flac_path)
    print("  写入后封面数量:", len(audio2.pictures))


def embed_cover_mp3(mp3_path, image_data):
    # 先确保有 ID3 tag
    try:
        id3 = ID3(mp3_path)
    except ID3NoHeaderError:
        id3 = ID3()

    # 删除旧封面（APIC）
    for key in list(id3.keys()):
        if key.startswith("APIC"):
            del id3[key]

    # 添加新封面
    id3.add(
        APIC(
            encoding=3,        # 3 = UTF-8
            mime="image/jpeg", # 你的链接是 jpg
            type=3,            # 3 = front cover
            desc="Cover",
            data=image_data
        )
    )

    id3.save(mp3_path)

    # 验证一下
    audio = MP3(mp3_path)
    apics = audio.tags.getall("APIC") if audio.tags else []
    print("  写入后封面数量:", len(apics))


for file in os.listdir(MUSIC_DIR):
    path = os.path.join(MUSIC_DIR, file)

    if file.lower().endswith(".flac"):
        audio = FLAC(path)
        artist = audio.get("artist", [""])[0]
        title = audio.get("title", [""])[0]

        print("\n处理(FLAC):", artist, "-", title)

        album_mid = search_qq_album_mid(artist, title)
        if not album_mid:
            print("  ❌ 找不到 albumMid")
            continue

        image_data = download_cover(album_mid)
        if not image_data:
            print("  ❌ 下载失败")
            continue

        embed_cover_flac(path, image_data)
        time.sleep(0.8)

    elif file.lower().endswith(".mp3"):
        audio = MP3(path)
        tags = audio.tags

        # MP3 的 artist/title 可能在 ID3 里：TPE1 / TIT2
        artist = ""
        title = ""
        if tags:
            if "TPE1" in tags and tags["TPE1"].text:
                artist = tags["TPE1"].text[0]
            if "TIT2" in tags and tags["TIT2"].text:
                title = tags["TIT2"].text[0]

        print("\n处理(MP3):", artist, "-", title)

        album_mid = search_qq_album_mid(artist, title)
        if not album_mid:
            print("  ❌ 找不到 albumMid")
            continue

        image_data = download_cover(album_mid)
        if not image_data:
            print("  ❌ 下载失败")
            continue

        embed_cover_mp3(path, image_data)
        time.sleep(0.8)

print("\n全部完成")

我用的 Hexo+Argon 搭建的。本地之前用 Typora，现在用 Obsidian 书写博客。

有一些要解决的东西，比如 Admonition 语法支持，新建 script 文件然后使用 admonition.js：

'use strict';

const TYPES = ['NOTE', 'TIP', 'IMPORTANT', 'WARNING', 'CAUTION'];

hexo.extend.filter.register('before_post_render', function (data) {
  data.content = data.content.replace(
    /^> \[!(\w+)\][\s\S]*?(?=\n(?!>)|\Z)/gm,
    function (block) {
      const lines = block.split('\n');
      const typeLine = lines.shift();
      const m = typeLine.match(/^> \[!(\w+)\]/);
      if (!m) return block;
      const type = m[1].toUpperCase();
      if (!TYPES.includes(type)) return block;

      const content = lines.map(line => line.replace(/^> ?/, '')).join('\n');

      return `<div class="admonition ${type.toLowerCase()}">
  <p class="admonition-title">${type}</p>
  <div class="admonition-content">${hexo.render.renderSync({ text: content, engine: 'markdown' })}</div>
</div>`;
    }
  );
  return data;
});

使用 feed-filter.js 添加对于 hexo-blog-encrypt 与 hexo-generator-feed 冲突（表现为加密博客显示了摘要）：

'use strict';

hexo.extend.filter.register('after_generate', async function () {
  const config = hexo.config;
  const route = hexo.route;
  
  const feedConfig = config.feed;
  if (!feedConfig || !feedConfig.path) {
    return;
  }
  
  const feedFiles = Array.isArray(feedConfig.path) 
    ? feedConfig.path 
    : [feedConfig.path];
  
  const encryptConfig = config.encrypt || {};
  const abstractText = encryptConfig.abstract || '该文章已加密，请前往原站查看。';
  
  const encryptedTitles = [];
  hexo.locals.get('posts').forEach(post => {
    if (post.password || 
        (post.encrypt && post.encrypt.password) ||
        post.enc_password) {
      encryptedTitles.push(post.title);
    }
  });

  if (encryptedTitles.length === 0) {
    return;
  }

  hexo.log.info(`[feed-encrypt-fix] 发现 ${encryptedTitles.length} 篇加密文章`);

  // 并行处理所有 feed 文件
  await Promise.all(feedFiles.map(async feedFile => {
    const stream = route.get(feedFile);
    if (!stream) {
      return;
    }

    // 读取 stream 内容
    const content = await new Promise((resolve, reject) => {
      const chunks = [];
      stream.on('data', chunk => chunks.push(chunk));
      stream.on('end', () => resolve(Buffer.concat(chunks).toString('utf-8')));
      stream.on('error', reject);
    });

    let modified = content;
    let hasChange = false;

    encryptedTitles.forEach(title => {
      const escaped = escapeRegExp(title);

      // Atom entry
      const atomPattern = new RegExp(
        `(<entry>(?:(?!<entry>)[\\s\\S])*?<title[^>]*>(?:<!\\[CDATA\\[)?\\s*${escaped}\\s*(?:\\]\\]>)?<\\/title>[\\s\\S]*?<\\/entry>)`,
        'g'
      );

      // RSS item
      const rssPattern = new RegExp(
        `(<item>(?:(?!<item>)[\\s\\S])*?<title[^>]*>(?:<!\\[CDATA\\[)?\\s*${escaped}\\s*(?:\\]\\]>)?<\\/title>[\\s\\S]*?<\\/item>)`,
        'g'
      );

      [atomPattern, rssPattern].forEach(pattern => {
        modified = modified.replace(pattern, function (match) {
          hasChange = true;
          
          match = match.replace(
            /<summary[^>]*>[\s\S]*?<\/summary>/gi,
            `<summary>${abstractText}</summary>`
          );
          
          match = match.replace(
            /<content[^>]*>[\s\S]*?<\/content>/gi,
            `<content type="html">${abstractText}</content>`
          );
          
          match = match.replace(
            /<description>[\s\S]*?<\/description>/gi,
            `<description><![CDATA[${abstractText}]]></description>`
          );
          
          return match;
        });
      });
    });

    if (hasChange) {
      route.set(feedFile, modified);
      hexo.log.info(`[feed-encrypt-fix] 已处理: ${feedFile}`);
    }
  }));
});

function escapeRegExp(string) {
  return string.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
}

使用 strip.js 添加对于部分主题文章摘要更好的渲染（避免 LaTeX 渲染 3 次或 bilibili-card 显示源码等情况）：

// scripts/strip.js
const cheerio = require('cheerio');

hexo.extend.filter.register('after_post_render', function(post){
  if (!post || !post.content) return post;

  const $ = cheerio.load(post.content);

  // 移除不想出现在 meta 的内容
  $(".bilibili-video, script, style").remove();
  $(".katex-html").remove();
  $('annotation[encoding="application/x-tex"]').remove();

  // 移除 Hexo {% ... %} 标签
  let text = $.html().replace(/{%\s*.*?\s*%}/gi, '');
  text = $(text).text().trim();

  // 截取摘要长度
  if (text.length > 160) text = text.substr(0, 160) + '…';

  // ⚠️ 写入 post.description，open_graph 会用它生成 meta
  post.description = text;

  return post;
});

OI 选手模拟赛要频繁测样例，可以用一个简单的脚本实现，例如在 Linux 环境下，在 /usr/local/bin 创建一个 run.sh，内容为：

#!/usr/bin/env bash
set -uo pipefail

# =====================================
# CP Local Runner (WSL2)
# Strict OJ (cgroup v2) + Peak RSS + Progress + Diagnostics
# CLI: run -SOURCE=xxx -TEST_PREFIX=roast ...
# =====================================

# -------- defaults (可被命令行覆盖) --------
SOURCE=""                 # 必填：不写 .cpp
TEST_PREFIX="a"           # 默认前缀
TIME_LIMIT=2000           # ms
MEM_LIMIT_MB=1024         # MB

TIME_ONLY=false
USE_SPJ=false
SPJ_EXEC="./spj"

SHOW_PROGRESS=true

CXX="g++"
CXXFLAGS="-O2 -std=c++14 -static" # 需要静态再加 -static
# -----------------------------------------

GREEN='\033[0;32m'
RED='\033[0;31m'
YELLOW='\033[1;33m'
NC='\033[0m'

usage() {
  cat <<'EOF'
Usage:
  run -SOURCE=a -TEST_PREFIX=a [options]

Options:
  -SOURCE <name>            Required. Source base name without .cpp
  -TEST_PREFIX <prefix>     Input file prefix, e.g. a => a1.in
  -TIME_LIMIT <ms>          Time limit in ms (default: 2000)
  -MEM_LIMIT_MB <mb>        Memory limit in MB (default: 1024)
  -TIME_ONLY <true|false>   Only measure time, skip judging
  -USE_SPJ <true|false>     Use SPJ
  -SPJ_EXEC <path>          SPJ executable path (default: ./spj)
  -SHOW_PROGRESS <true|false>
  -CXX <compiler>           default: g++
  -CXXFLAGS <flags>         quote it, e.g. -CXXFLAGS "-O2 -std=c++14 -static"

Examples:
  run -SOURCE=a -TEST_PREFIX=a
  run -SOURCE a -TEST_PREFIX a -TIME_LIMIT 10000 -MEM_LIMIT_MB 256
EOF
}

# -------- parse args --------
# 支持：-KEY=VAL 和 -KEY VAL
for arg in "$@"; do
  case "$arg" in
    -h|--help) usage; exit 0 ;;
  esac
done

i=1
while (( i <= $# )); do
  a="${!i}"

  if [[ "$a" == -*=* ]]; then
    key="${a%%=*}"
    val="${a#*=}"
    case "$key" in
      -SOURCE) SOURCE="$val" ;;
      -TEST_PREFIX) TEST_PREFIX="$val" ;;
      -TIME_LIMIT) TIME_LIMIT="$val" ;;
      -MEM_LIMIT_MB) MEM_LIMIT_MB="$val" ;;
      -TIME_ONLY) TIME_ONLY="$val" ;;
      -USE_SPJ) USE_SPJ="$val" ;;
      -SPJ_EXEC) SPJ_EXEC="$val" ;;
      -SHOW_PROGRESS) SHOW_PROGRESS="$val" ;;
      -CXX) CXX="$val" ;;
      -CXXFLAGS) CXXFLAGS="$val" ;;
      *) echo -e "${RED}Unknown option: $key${NC}"; usage; exit 1 ;;
    esac
    ((i++))
  else
    key="$a"
    ((i++))
    val="${!i-}"
    case "$key" in
      -SOURCE) SOURCE="$val" ;;
      -TEST_PREFIX) TEST_PREFIX="$val" ;;
      -TIME_LIMIT) TIME_LIMIT="$val" ;;
      -MEM_LIMIT_MB) MEM_LIMIT_MB="$val" ;;
      -TIME_ONLY) TIME_ONLY="$val" ;;
      -USE_SPJ) USE_SPJ="$val" ;;
      -SPJ_EXEC) SPJ_EXEC="$val" ;;
      -SHOW_PROGRESS) SHOW_PROGRESS="$val" ;;
      -CXX) CXX="$val" ;;
      -CXXFLAGS) CXXFLAGS="$val" ;;
      *)
        echo -e "${RED}Unknown option: $key${NC}"
        usage
        exit 1
        ;;
    esac
    ((i++))
  fi
done

if [[ -z "${SOURCE:-}" ]]; then
  echo -e "${RED}-SOURCE is required.${NC}"
  usage
  exit 1
fi

# -------- hints (Diagnostics) --------
run_hint() {
  local run="$1"
  case "$run" in
    TLE) echo "time limit (algo/IO/loop)" ;;
    MLE) echo "memory limit (big array/recursion)" ;;
    RE\(*SIGSEGV*\)) echo "segfault (OOB/null/stack)" ;;
    RE\(*SIGABRT*\)) echo "abort/assert failed" ;;
    RE\(*SIGFPE*\)) echo "divide by 0 / arithmetic" ;;
    RE\(exit=*\)) echo "non-zero exit" ;;
    RE\(*SIGKILL*\)) echo "killed (maybe OOM/external kill)" ;;
    *) echo "" ;;
  esac
}
judge_hint() {
  local judge="$1"
  case "$judge" in
    WA) echo "output mismatch (format/edge)" ;;
    NO_ANS) echo "missing *.out/*.ans" ;;
    *) echo "" ;;
  esac
}

# -------- utils --------
ms_to_timeout_arg() { printf "%d.%03ds" $((TIME_LIMIT/1000)) $((TIME_LIMIT%1000)); }

signal_name_from_code() {
  local code="$1"
  (( code < 128 )) && { echo ""; return; }
  local sig=$((code - 128))
  case "$sig" in
    6) echo "SIGABRT" ;;
    7) echo "SIGFPE" ;;
    8) echo "SIGKILL" ;;
    9) echo "SIGSEGV" ;;
    10) echo "SIGTERM" ;;
    *) echo "SIG${sig}" ;;
  esac
}

bytes_to_mib_1dp() { awk -v b="$1" 'BEGIN{printf "%.1f", b/1024/1024}'; }

cgroup_available() { [[ -f /sys/fs/cgroup/cgroup.controllers ]]; }

cgroup_setup() {
  local mem_bytes="$1"
  local cg="/sys/fs/cgroup/cp_runner_${SUDO_USER:-$USER}_$$"
  mkdir "$cg" 2>/dev/null || return 1
  echo "$mem_bytes" > "$cg/memory.max" 2>/dev/null || return 2
  echo 0 > "$cg/memory.peak" 2>/dev/null || true
  echo "$cg"
}
cgroup_cleanup() { rmdir "$1" 2>/dev/null || true; }

cgroup_read_kv() { awk -v k="$2" '$1==k {print $2}' "$1" 2>/dev/null || echo ""; }
cgroup_reset_peak() { echo 0 > "$1/memory.peak" 2>/dev/null || true; }
cgroup_get_peak_bytes() { cat "$1/memory.peak" 2>/dev/null || echo "0"; }

# -------- progress --------
PROGRESS_PID=""
start_progress() {
  local label="$1"
  $SHOW_PROGRESS || return 0
  (
    local t=0
    while true; do
      awk -v tenths="$t" -v lab="$label" 'BEGIN{
        printf("\rRunning %-15s  Elapsed: %.1fs", lab, tenths/10.0);
        fflush();
      }' 1>&2
      t=$((t+1))
      sleep 0.1
    done
  ) &
  PROGRESS_PID=$!
}
stop_progress() {
  $SHOW_PROGRESS || return 0
  if [[ -n "${PROGRESS_PID:-}" ]]; then
    kill "$PROGRESS_PID" 2>/dev/null || true
    wait "$PROGRESS_PID" 2>/dev/null || true
    PROGRESS_PID=""
    printf "\r%*s\r" 80 "" 1>&2
  fi
}

# run_one: status time_ms run_reason peak_bytes oom_kill_delta
run_one() {
  local cg="$1" infile="$2" tmp_out="$3"
  local t_arg; t_arg="$(ms_to_timeout_arg)"
  local start_ns end_ns time_used status sig reason
  local oom_before oom_after oom_delta peak

  oom_before="$(cgroup_read_kv "$cg/memory.events" "oom_kill")"; [[ -z "$oom_before" ]] && oom_before=0
  cgroup_reset_peak "$cg"

  start_ns=$(date +%s%N)
  start_progress "$infile"

  ( timeout "$t_arg" bash -c "exec $EXEC < '$infile' > '$tmp_out' 2>/dev/null" ) &
  local pid=$!
  echo "$pid" > "$cg/cgroup.procs" 2>/dev/null || true
  wait "$pid"
  status=$?

  end_ns=$(date +%s%N)
  time_used=$(( (end_ns - start_ns) / 1000000 ))

  stop_progress

  oom_after="$(cgroup_read_kv "$cg/memory.events" "oom_kill")"; [[ -z "$oom_after" ]] && oom_after="$oom_before"
  oom_delta=$((oom_after - oom_before))
  peak="$(cgroup_get_peak_bytes "$cg")"; [[ -z "$peak" ]] && peak=0

  if [[ $status -eq 124 ]]; then
    reason="TLE"
  elif (( oom_delta > 0 )); then
    reason="MLE"
  else
    sig="$(signal_name_from_code "$status")"
    if [[ -n "$sig" ]]; then
      reason="RE(${sig})"
    elif [[ $status -ne 0 ]]; then
      reason="RE(exit=$status)"
    else
      reason="OK"
    fi
  fi

  echo "$status $time_used $reason $peak $oom_delta"
}

SRC_FILE="${SOURCE}.cpp"
EXEC="./${SOURCE}"

if [[ ! -f "$SRC_FILE" ]]; then
  echo -e "${RED}Source file $SRC_FILE not found in current directory.${NC}"
  exit 1
fi

echo "Compiling $SRC_FILE ..."
$CXX $CXXFLAGS "$SRC_FILE" -o "${EXEC#./}"
if [[ $? -ne 0 ]]; then
  echo -e "${RED}Compile Error${NC}"
  exit 1
fi
echo -e "${GREEN}Compile Success${NC}"

INPUT_REGEX="^(ex_)?${TEST_PREFIX}[0-9]+\.in$"
mapfile -t INPUT_FILES < <(ls 2>/dev/null | grep -E "$INPUT_REGEX" | sort -V)
if [[ ${#INPUT_FILES[@]} -eq 0 ]]; then
  echo "No input files found."
  exit 1
fi

if ! cgroup_available; then
  echo -e "${RED}cgroup v2 not available.${NC}"
  exit 1
fi

CGROUP_PATH="$(cgroup_setup $((MEM_LIMIT_MB*1024*1024)) 2>/dev/null || true)"
if [[ -z "$CGROUP_PATH" ]]; then
  echo -e "${RED}Cannot create cgroup. For strict OJ limits, run:${NC} sudo run ..."
  exit 1
fi

printf "%-15s %-10s %-12s %-10s %-10s\n" "Testcase" "Time(ms)" "Peak(MiB)" "Run" "Judge"
printf "%-15s %-10s %-12s %-10s %-10s\n" "--------" "--------" "--------" "--------" "--------"

TOTAL=0
PASS=0

declare -a DIAG_CASE DIAG_RUN DIAG_JUDGE DIAG_MSG
DIAG_CASE=(); DIAG_RUN=(); DIAG_JUDGE=(); DIAG_MSG=()

for infile in "${INPUT_FILES[@]}"; do
  ((TOTAL++))
  base="${infile%.in}"

  ans=""
  [[ -f "${base}.out" ]] && ans="${base}.out"
  [[ -z "$ans" && -f "${base}.ans" ]] && ans="${base}.ans"

  tmp_out="tmp_${base}.out"
  rm -f "$tmp_out" 2>/dev/null || true

  read -r status time_used run_reason peak_bytes oom_delta < <(run_one "$CGROUP_PATH" "$infile" "$tmp_out")
  peak_mib="$(bytes_to_mib_1dp "${peak_bytes:-0}")"

  if [[ "$run_reason" != "OK" ]]; then
    printf "%-15s %-10s %-12s ${RED}%-10s${NC} %-10s\n" \
      "$infile" "$time_used" "$peak_mib" "$run_reason" "-"

    msg="$(run_hint "$run_reason")"
    DIAG_CASE+=("$infile"); DIAG_RUN+=("$run_reason"); DIAG_JUDGE+=("-"); DIAG_MSG+=("${msg:-runtime failure}")

    rm -f "$tmp_out" 2>/dev/null || true
    continue
  fi

  if $TIME_ONLY; then
    ((PASS++))
    printf "%-15s %-10s %-12s ${YELLOW}%-10s${NC} %-10s\n" \
      "$infile" "$time_used" "$peak_mib" "TIME" "-"
    rm -f "$tmp_out" 2>/dev/null || true
    continue
  fi

  judge="-"
  judge_color=$GREEN
  if $USE_SPJ; then
    if [[ -z "$ans" ]]; then
      judge="NO_ANS"; judge_color=$YELLOW
    else
      $SPJ_EXEC "$infile" "$tmp_out" "$ans"
      if [[ $? -eq 0 ]]; then
        judge="AC"; judge_color=$GREEN; ((PASS++))
      else
        judge="WA"; judge_color=$RED
      fi
    fi
  else
    if [[ -z "$ans" ]]; then
      judge="NO_ANS"; judge_color=$YELLOW
    elif diff -Z -B "$tmp_out" "$ans" >/dev/null; then
      judge="AC"; judge_color=$GREEN; ((PASS++))
    else
      judge="WA"; judge_color=$RED
    fi
  fi

  printf "%-15s %-10s %-12s ${GREEN}%-10s${NC} ${judge_color}%-10s${NC}\n" \
    "$infile" "$time_used" "$peak_mib" "OK" "$judge"

  if [[ "$judge" != "AC" ]]; then
    msg="$(judge_hint "$judge")"
    DIAG_CASE+=("$infile"); DIAG_RUN+=("OK"); DIAG_JUDGE+=("$judge"); DIAG_MSG+=("${msg:-judge failed}")
  fi

  rm -f "$tmp_out" 2>/dev/null || true
done

echo
echo "======================================"
echo -e "Passed: ${PASS}/${TOTAL}"
echo "======================================"

if (( ${#DIAG_CASE[@]} > 0 )); then
  echo
  echo "Diagnostics:"
  for i in "${!DIAG_CASE[@]}"; do
    echo "- ${DIAG_CASE[$i]}: Run=${DIAG_RUN[$i]} Judge=${DIAG_JUDGE[$i]}  (${DIAG_MSG[$i]})"
  done
fi

cgroup_cleanup "$CGROUP_PATH"
